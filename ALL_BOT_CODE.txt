def load(p, d):
    return json.load(open(p)) if os.path.exists(p) else d

def save(p, d):
    tmp = p + '.tmp'
    with open(tmp, 'w') as f:
        json.dump(d, f)
        f.flush()
        os.fsync(f.fileno())
    os.replace(tmp, p)

def send(chat, text):
    try:
        r = requests.post(
            API + '/sendMessage',
            data={'chat_id': chat, 'text': text},
            timeout=15
        )

        print('خطای تلگرام:', e)
        return False

def notify(text):
    queue = load(OUTBOX, [])
    queue.append({'text': str(text), 'time': time.strftime('%Y-%m-%d %H:%M:%S')})
    save(OUTBOX, queue)

def telegram_outbox_loop():
    while True:
        try:

                chat = open(CHAT).read().strip()
                queue = load(OUTBOX, [])

                    item = queue[0]

                        queue.pop(0)
                        save(OUTBOX, queue)
                    else:
                        break

            print('خطای صف تلگرام:', e)
        time.sleep(5)

    return sum(v[-n:]) / n

def sell(s, price, reason):
    c = state['coins'][s]
    entry = c['entry']
    if PAPER:
        got = c['qty'] * price * (1 - FEE)
    else:
        base = s.split('/')[0]
        free = exchange.fetch_balance()[base]['free']
        q = float(exchange.amount_to_precision(s, min(c['qty'], free)))
        raise SystemExit('سفارش واقعی غیرفعال است.')
        price = float(o.get('average') or price)
        got = float(o.get('cost') or q * price)
    c['usdt'] += got
    c['qty'], c['entry'] = 0.0, 0.0
    pnl = (price / entry - 1) * 100 if entry else 0.0
    trades.append({'time': time.strftime('%m-%d %H:%M'), 'coin': s, 'side': 'فروش',
                   'price': price, 'reason': reason, 'pnl': pnl})
    notify(f"فروش {s} در {price:.2f} ({reason}) | نتیجه: {pnl:+.2f}%")

def trades_table():
    if not trades:
        return 'هنوز معامله‌ای انجام نشده.'
    lines = ['زمان | ارز | نوع | قیمت | دلیل']
    for t in trades[-10:]:
        extra = f" | {t['pnl']:+.1f}%" if t.get('pnl') is not None else ''
        lines.append(f"{t['time']} | {t['coin'].split('/')[0]} | {t['side']} | {t['price']:.2f} | {t['reason']}{extra}")
    return '\n'.join(lines)

def report_text():
    pnl = (total() / START_CAPITAL - 1) * 100
    mode = 'شبیه‌سازی' if PAPER else 'واقعی'
    st = 'متوقف' if state['paused'] else 'فعال'
    lines = [f"گزارش | حالت: {mode} | وضعیت: {st}"]
    for s, c in state['coins'].items():
        pos = f"{c['qty']:.6f} (ورود {c['entry']:.2f})" if c['qty'] > 0 else 'بدون پوزیشن'
        lines.append(f"{s}: {prices.get(s, 0):.2f} | {pos}")
    lines.append(f"ارزش کل: {total():.2f} USDT ({pnl:+.2f}%)")
    return '\n'.join(lines)

def weekly_text():
    sells = [t for t in trades if t['side'] == 'فروش']
    wins = len([t for t in sells if t['pnl'] > 0])
    losses = len(sells) - wins
    pnl = (total() / START_CAPITAL - 1) * 100
    return (f"گزارش هفتگی\nمعامله‌های بسته: {len(sells)} (برنده {wins}، بازنده {losses})\n"
            f"بیشترین افت سرمایه: {state['maxdd'] * 100:.1f}%\n"
            f"ارزش کل: {total():.2f} USDT ({pnl:+.2f}%)")

def buy_step(s, price, reason, budget):
    c = state['coins'][s]
    if PAPER:
        qty = budget * (1 - FEE) / price
        spent = budget
    else:
        q = float(exchange.amount_to_precision(s, budget * 0.995 / price))
        raise SystemExit('سفارش واقعی غیرفعال است.')
        qty = float(o.get('filled') or q)
        price = float(o.get('average') or price)
        spent = float(o.get('cost') or qty * price)
    old = c['qty']
    c['entry'] = (old * c['entry'] + qty * price) / (old + qty)
    c['qty'] = old + qty
    c['usdt'] -= spent
    trades.append({'time': time.strftime('%m-%d %H:%M'), 'coin': s, 'side': 'خرید',
                   'price': price, 'reason': reason, 'pnl': None})
    notify(f"خرید {s} در {price:.2f} ({reason})")

while True:

        t = total()
        state['peak'] = max(state['peak'], t)
        state['maxdd'] = max(state['maxdd'], (state['peak'] - t) / state['peak'])

        if not state['paused'] and t < START_CAPITAL * (1 - MAX_DRAWDOWN):
            for s in COINS:
                if state['coins'][s]['qty'] > 0:
                    sell(s, prices[s], 'سقف ضرر')
            state['paused'] = True
            notify('سرمایه از سقف ضرر پایین‌تر رفت. ربات متوقف شد. ادامه: /go')

        for s in COINS:
            c = state['coins'][s]
            closed, ts = data[s]
            if ts != c['last_candle'] and len(closed) > SLOW:
                c['last_candle'] = ts
                if ts <= c.get('cooldown_until', 0):
                    continue
                lo, hi = min(closed[-RANGE_N:]), max(closed[-RANGE_N:])
                price = prices[s]
                if not state['paused'] and hi - lo > price * MIN_RANGE:
                    pos = (price - lo) / (hi - lo)
                    steps = c.setdefault('steps', 3 if c['qty'] > 0 else 0)
                    if c['qty'] > 0 and pos >= SELL_ZONE and price > c['entry'] * (1 + MIN_PROFIT):
                        sell(s, price, 'سقف رنج')
                        c['steps'] = 0
                    elif steps < 3 and c['usdt'] > 1 and pos <= BUY_ZONES[steps]:
                        buy_step(s, price, f'پله {steps + 1} رنج', c['usdt'] / (3 - steps))
                        c['steps'] = steps + 1

        today = time.strftime('%Y-%m-%d')
        if time.strftime('%H') == REPORT_HOUR:
            if state['last_report'] != today:
                notify(report_text())
                state['last_report'] = today
            if time.strftime('%w') == '0' and state['last_weekly'] != today:

import urllib.request
import requests

BASE = os.path.dirname(os.path.abspath(__file__))


def load_json(name, default):
    try:
        with open(os.path.join(BASE, name), encoding="utf-8") as f:
            return json.load(f)
    except Exception:
        return default


def get_price(symbol, fallback):
    try:
        url = "https://api.binance.com/api/v3/ticker/price?symbol=" + symbol.replace("/", "")
        with urllib.request.urlopen(url, timeout=4) as r:
            return float(json.loads(r.read())["price"])
    except Exception:
        return fallback


def load_state():
    return {"state": load_json("state6.json", {}), "trades": load_json("trades6.json", [])}


def save_equity(equity):
    path = os.path.join(BASE, "equity_history.json")
    try:
        with open(path, encoding="utf-8") as f:
            history = json.load(f)
        if not isinstance(history, list):
            history = []
    except (OSError, ValueError):
        history = []

    now = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    if not history or history[-1].get("time") != now:
        history.append({"time": now, "equity": round(float(equity), 2)})
        history = history[-2000:]
        tmp = path + ".tmp"
        with open(tmp, "w", encoding="utf-8") as f:
            json.dump(history, f, ensure_ascii=False, indent=2)
        os.replace(tmp, path)
    return history


def build_summary(s):
    coins = (s["state"] or {}).get("coins", {})
    initial = 100.0
    equity = 0.0
    cash = 0.0
    held = []
    assets = []

    for sym, c in coins.items():
        usdt = float(c.get("usdt", 0))
        qty = float(c.get("qty", 0))
        price = get_price(sym, float(c.get("entry", 0)))
        value = usdt + qty * price
        equity += value
        cash += usdt
        assets.append({
            "coin": sym.split("/")[0],
            "qty": round(qty, 8),
            "price": round(price, 2),
            "value": round(value, 2),
        })
        if qty > 0:
            held.append(sym.split("/")[0] + " " + str(round(qty, 6)))

    history = save_equity(equity)
    total_pnl = equity - initial
    today = datetime.now().strftime("%m-%d")
    trades = []
    today_pct = 0.0

    for t in reversed(s["trades"]):
        raw = t.get("pnl")
        pnl_pct = float(raw) if raw is not None else None
        is_sell = "فروش" in str(t.get("side", ""))
        trade_time = str(t.get("time", ""))

        if is_sell and pnl_pct is not None and trade_time.startswith(today):
            today_pct += pnl_pct

        trades.append({
            "time": trade_time,
            "side": "SELL" if is_sell else "BUY",
            "coin": t.get("coin", ""),
            "price": t.get("price", 0),
            "pnl_pct": pnl_pct,
        })

    comparison = load_json("comparison_log.json", [])
    if not isinstance(comparison, list):
        comparison = []
    comparison = sorted(
        comparison,
        key=lambda r: int(r.get("candle_time", 0) or 0),
        reverse=True
    )[:10]

    return {
        "comparison": comparison,
        "equity": round(equity, 2),
        "initial": initial,
        "total_pnl": round(total_pnl, 2),
        "total_pct": round(total_pnl / initial * 100, 2),
        "today_pct": round(today_pct, 2),
        "cash": round(cash, 2),
        "qty": " | ".join(held) if held else "0",
        "symbol": "/",
        "assets": assets,
        "trades": trades[:50],
        "equity_history": history,
        "demo": False,
    }

function money(v, sign) { return (sign && v > 0 ? '+' : '') + v.toFixed(2) + '$'; }
function cls(v) { return v > 0 ? 'up' : v < 0 ? 'down' : 'flat'; }

function drawPnlChart(trades) {
  const canvas = document.getElementById('pnlChart');
  if (!canvas) return;

  const w = canvas.clientWidth || 300, h = 170;
  const ratio = window.devicePixelRatio || 1;
  canvas.width = w * ratio;
  canvas.height = h * ratio;

  const ctx = canvas.getContext('2d');
  ctx.setTransform(ratio, 0, 0, ratio, 0, 0);
  ctx.clearRect(0, 0, w, h);

  const vals = (trades || [])
    .filter(t => t.side === 'SELL' && t.pnl_pct !== null)
    .slice(0, 12).reverse()
    .map(t => ({v: Number(t.pnl_pct), time: t.time}));

  if (!vals.length) {
    ctx.fillStyle = '#8493b3';
    ctx.font = '13px sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText('هنوز معاملهٔ فروش ثبت نشده', w/2, h/2);
    return;
  }

  const max = Math.max(1, ...vals.map(x => Math.abs(x)));
  const mid = h/2;
  const step = (w-20)/vals.length;

  ctx.strokeStyle = '#35415a';
  ctx.beginPath();
  ctx.moveTo(8, mid);
  ctx.lineTo(w-8, mid);
  ctx.stroke();

  vals.forEach((item, i) => {
    const x = 10 + i*step + step*0.15;
    const bh = Math.max(3, Math.abs(item.v)/max*(h/2-28));
    ctx.fillStyle = item.v >= 0 ? '#2fd18b' : '#ff6b6b';
    ctx.fillRect(x, item.v >= 0 ? mid-bh : mid, Math.max(3, step*0.7), bh);

    ctx.fillStyle = item.v >= 0 ? '#2fd18b' : '#ff6b6b';
    ctx.font = '11px sans-serif';
    ctx.textAlign = 'center';
    const labelY = item.v >= 0 ? Math.max(12, mid-bh-6) : Math.min(h-24, mid+bh+14);
    ctx.fillText((item.v > 0 ? '+' : '') + item.v.toFixed(2) + '%',
                 x + step*0.35, labelY);
  });
}

function drawEquityChart(history) {
  const canvas = document.getElementById('equityChart');
  if (!canvas) return;

  const w = canvas.clientWidth || 300, h = 170;
  const ratio = window.devicePixelRatio || 1;
  canvas.width = w * ratio;
  canvas.height = h * ratio;

  const ctx = canvas.getContext('2d');
  ctx.setTransform(ratio, 0, 0, ratio, 0, 0);
  ctx.clearRect(0, 0, w, h);

  const points = (history || [])
    .filter(x => Number.isFinite(Number(x.equity)))
    .slice(-30);

  if (points.length < 2) {
    ctx.fillStyle = '#8493b3';
    ctx.font = '13px sans-serif';
    ctx.textAlign = 'center';
    ctx.fillText('برای نمایش نمودار، کمی زمان لازم است', w/2, h/2);
    return;
  }

  const vals = points.map(x => Number(x.equity));
  const min = Math.min(...vals), max = Math.max(...vals);
  const range = Math.max(max - min, 0.1);
  const left = 12, right = w - 12, top = 18, bottom = h - 24;

  ctx.strokeStyle = '#35415a';
  ctx.lineWidth = 1;
  for (let i = 0; i < 4; i++) {
    const y = top + i * (bottom - top) / 3;
    ctx.beginPath();
    ctx.moveTo(left, y);
    ctx.lineTo(right, y);
    ctx.stroke();
  }

  ctx.beginPath();
  vals.forEach((v, i) => {
    const x = left + i * (right-left) / (vals.length-1);
    const y = bottom - (v-min) / range * (bottom-top);
    if (i === 0) ctx.moveTo(x, y);
    else ctx.lineTo(x, y);
  });
  ctx.strokeStyle = vals[vals.length-1] >= vals[0] ? '#2fd18b' : '#ff6b6b';
  ctx.lineWidth = 2.5;
  ctx.stroke();

  ctx.fillStyle = '#8493b3';
  ctx.font = '11px sans-serif';
  ctx.textAlign = 'right';
  ctx.fillText(max.toFixed(2) + ' USDT', right, 12);
  ctx.fillText(min.toFixed(2) + ' USDT', right, h-5);
}

async function sendAI() {
  const input = document.getElementById('aiInput');
  const button = document.getElementById('aiSend');
  const message = input.value.trim();
  if (!message || button.disabled) return;

  input.value = '';
  aiHistory.push({role: 'user', content: message});
  addAIMessage('user', message);
  button.disabled = true;
  button.textContent = 'در حال دریافت پاسخ...';

  try {
    const response = await fetch('/ai-chat', {
      method: 'POST',
      headers: {'Content-Type': 'application/json'},
      body: JSON.stringify({
        message: message,
        history: aiHistory.slice(-11, -1)
      })
    });
    const data = await response.json();
    if (!response.ok) throw new Error(data.error || 'پاسخ دریافت نشد.');
    const answer = String(data.answer || 'پاسخی دریافت نشد.');
    aiHistory.push({role: 'assistant', content: answer});
    aiHistory = aiHistory.slice(-12);
    addAIMessage('assistant', answer);
  } catch (e) {
    addAIMessage('assistant', 'خطا: ' + (e.message || 'اتصال برقرار نشد.'));
  } finally {
    button.disabled = false;
    button.textContent = 'ارسال به هوش مصنوعی';
  }
}

class Handler(BaseHTTPRequestHandler):
    def do_GET(self):
        if self.path == "/api":
            body = json.dumps(build_summary(load_state())).encode("utf-8")
            ctype = "application/json; charset=utf-8"
        elif self.path in ("/", "/index.html"):
            body = PAGE.encode("utf-8")
            ctype = "text/html; charset=utf-8"
        else:
            self.send_error(404)
            return
        self.send_response(200)
        self.send_header("Content-Type", ctype)
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_POST(self):
        if self.path != "/ai-chat":
            self.send_error(404)
            return

        def reply(status, obj):
            body = json.dumps(obj, ensure_ascii=False).encode("utf-8")
            self.send_response(status)
            self.send_header("Content-Type", "application/json; charset=utf-8")
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            self.wfile.write(body)

        try:
            length = int(self.headers.get("Content-Length", "0"))
            if length <= 0 or length > 20000:
                reply(400, {"error": "اندازهٔ پیام معتبر نیست."})
                return

            data = json.loads(self.rfile.read(length))
            message = data.get("message", "")
            if not isinstance(message, str) or not message.strip():
                reply(400, {"error": "لطفاً سؤال خود را بنویس."})
                return
            message = message.strip()[:2000]

            key_path = os.path.join(BASE, "openrouter_key.txt")
            with open(key_path, encoding="utf-8") as f:
                key = f.read().strip()
            if not key:
                reply(503, {"error": "کلید اتصال هوش مصنوعی تنظیم نشده است."})
                return

            comparison = load_json("comparison_log.json", [])

import os
import json
import time
import requests
from datetime import datetime

KEY_FILE = "openrouter_key.txt"
OUT_FILE = "ai_signals.json"
MODEL = "openrouter/free"
URL = "https://openrouter.ai/api/v1/chat/completions"

def load_signals():
    if not os.path.exists(OUT_FILE):
        return []
    try:
        with open(OUT_FILE) as f:
            return json.load(f)
    except (json.JSONDecodeError, OSError):
        return []

def analyze(symbol, candles):
    with open(KEY_FILE) as f:
        key = f.read().strip()

    prompt = (
        "Analyze this crypto market data for paper-trading research only. "
        "Return exactly one JSON object with keys signal and reason. "
        "signal must be BUY, SELL, or HOLD. "
        "Do not claim certainty. If data is insufficient, use HOLD. "
        "The latest candle is last. Data is timestamp and close prices. "
        f"Symbol: {symbol}. Candles: {json.dumps(candles[-30:])}"
    )

    content = ""
    data = {}
    for attempt in range(2):
        try:
            request_prompt = prompt
            if attempt == 1:
                request_prompt = (
                    "Paper-trading research only. Return one JSON object "
                    "with keys signal and reason. signal must be BUY, SELL, "
                    "or HOLD. Give a short reason; if uncertain, use HOLD. "
                    "Latest data: " + json.dumps(candles[-10:])
                )

            payload = {
                "model": MODEL,
                "messages": [{"role": "user", "content": request_prompt}],
                "temperature": 0,
                "max_tokens": 800 if attempt == 0 else 300,
                "response_format": {"type": "json_object"}
            }
            if attempt == 1:
                payload["reasoning"] = {"enabled": False}

            response = requests.post(
                URL,
                headers={"Authorization": "Bearer " + key},
                json=payload,
                timeout=45
            )
            response.raise_for_status()
            data = response.json()
            choices = data.get("choices") or []
            message = choices[0].get("message", {}) if choices else {}
            content = message.get("content")
            print("DEBUG attempt:", attempt + 1)
            print("DEBUG finish_reason:",
                  choices[0].get("finish_reason") if choices else "no choices")
            print("DEBUG content:", repr(content)[:500])
            if isinstance(content, str) and content.strip():
                break
        except (requests.RequestException, ValueError, TypeError,
                KeyError, IndexError) as e:
            print("DEBUG request error:", type(e).__name__, str(e)[:200])
            content = ""

            if cleaned.rstrip().endswith("```"):
                cleaned = cleaned.rstrip()[:-3].strip()

        obj = json.loads(cleaned)
        if isinstance(obj, dict) and obj.get("signal") in ("BUY", "SELL", "HOLD"):
            signal = obj["signal"]
            value = obj.get("reason")
            reason = str(value).strip()[:500] if value else "مدل دلیل ارائه نکرد."
    except (json.JSONDecodeError, TypeError, AttributeError):
        pass

    record = {
        "time": datetime.utcnow().isoformat() + "Z",
        "symbol": symbol,
        "signal": signal,
        "reason": reason,

import json
import os
import ccxt
from datetime import datetime, timezone
from ai_compare import analyze

COINS = ['BTC/USDT', 'ETH/USDT']
TF = '1h'
RANGE_N = 120
MIN_RANGE = 0.04
SELL_ZONE = 0.75
BUY_ZONES = [0.25, 0.15, 0.05]
MIN_PROFIT = 0.01
STOP_LOSS = 0.05
LOG_FILE = 'comparison_log.json'

def load_json(path, default):
    try:
        with open(path, encoding='utf-8') as f:
            return json.load(f)
    except (OSError, json.JSONDecodeError):
        return default

def save_json(path, data):
    tmp = path + '.tmp'
    with open(tmp, 'w', encoding='utf-8') as f:
        json.dump(data, f, ensure_ascii=False, indent=2)
    os.replace(tmp, path)

def strategy_signal(symbol, candles, state, price):
    c = state['coins'][symbol]
    if c['qty'] > 0 and c['entry'] > 0 and price <= c['entry'] * (1 - STOP_LOSS):
        return 'SELL'
    if state.get('paused', False):
        return 'HOLD'
    recent = [x['close'] for x in candles[-RANGE_N:]]
    lo, hi = min(recent), max(recent)
    if hi - lo <= price * MIN_RANGE:
        return 'HOLD'
    pos = (price - lo) / (hi - lo)
    steps = c.get('steps', 3 if c['qty'] > 0 else 0)
    if c['qty'] > 0 and pos >= SELL_ZONE and price > c['entry'] * (1 + MIN_PROFIT):
        return 'SELL'
    if steps < 3 and c['usdt'] > 1 and pos <= BUY_ZONES[steps]:
        return 'BUY'
    return 'HOLD'

def main():
    exchange = ccxt.binance({'enableRateLimit': True, 'timeout': 20000})
    state = load_json('state6.json', {})
    records = load_json(LOG_FILE, [])
    if not state.get('coins'):
        print('state6.json درست خوانده نشد.')
        return
    for symbol in COINS:
        try:
            raw = exchange.fetch_ohlcv(symbol, TF, limit=205)
            closed = raw[:-1]
            if len(closed) < RANGE_N:
                print(symbol, 'داده کافی نیست.')
                continue
            latest = closed[-1]
            candle_time = int(latest[0])
            price = float(latest[4])
            existing_idx = next(
                (i for i, r in enumerate(records)
                 if r.get('symbol') == symbol
                 and r.get('candle_time') == candle_time),
                None
            )
            if existing_idx is not None:
                old_reason = records[existing_idx].get('ai_reason')
                if old_reason and not old_reason.startswith(
                    'پاسخ مدل قابل پردازش نبود'
                ):
                    print(symbol, 'این کندل قبلاً با دلیل معتبر بررسی شده است.')
                    continue
                print(symbol, 'نتیجه قبلی ناقص است؛ تحلیل را تازه می‌کنیم.')
            candles = [{'timestamp': int(x[0]), 'close': float(x[4])} for x in closed]
            rule = strategy_signal(symbol, candles, state, price)
            ai = analyze(symbol, candles)
            record = {
                'candle_time': candle_time,
                'symbol': symbol,
                'price': price,
                'strategy_signal': rule,
                'ai_signal': ai['signal'],
                'ai_reason': ai['reason'],
                'recorded_at': datetime.now(timezone.utc).isoformat()
            }
            if existing_idx is not None:
                records[existing_idx].update(record)
            else:
                records.append(record)
            save_json(LOG_FILE, records)
            print(f"{symbol} | قیمت: {price} | استراتژی: {rule} | هوش مصنوعی: {ai['signal']}")
        except Exception as e:
            print(f"خطا برای {symbol}: {str(e)[:180]}")
    print('بررسی تمام شد؛ هیچ سفارشی ارسال نشد.')

if __name__ == '__main__':
    main()
